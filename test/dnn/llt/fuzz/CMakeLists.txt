cmake_minimum_required(VERSION 3.14)

project(KUDNN_FUZZ LANGUAGES C CXX)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_compile_options(-fPIC -Wall -Werror -Wpedantic -fopenmp -Wextra -Werror -march=armv8.5-a+bf16 -fsanitize-coverage=trace-pc)
if (${COMP_TOOLCHAIN} STREQUAL "clang")
    add_compile_options(-Wno-gnu-zero-variadic-macro-arguments)
endif ()

if (NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    add_compile_options(-fstack-protector-strong -O3)
    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -s")
else ()
    add_compile_options(-O1)
endif ()

include("${PROJECT_SOURCE_DIR}/cmake/FindKuDNN.cmake")
if (KUDNN_FOUND)
    include_directories(${KUDNN_INCLUDE_DIRS})
    message(STATUS "FUZZ Library: ${KUDNN_LIBRARIES}")
    message(STATUS "FUZZ Library headers: ${KUDNN_INCLUDE_DIRS}")
endif ()

if (${TARGET_PLATFORM} STREQUAL "KP920F")
    add_compile_options(-DKUDNNL_920F)
elseif(${TARGET_PLATFORM} STREQUAL "KP920B")
    add_compile_options(-DKUDNNL_920B)
elseif(${TARGET_PLATFORM} STREQUAL "KP920G")
    add_compile_options(-DKUDNNL_920G)
endif()

# header directories
include_directories(${AARCH64_CODEGEN_PATH})
include_directories(${PROJECT_SOURCE_DIR}/include)
include_directories(${PROJECT_SOURCE_DIR}/src)
link_directories(${PROJECT_SOURCE_DIR}/lib)

# add sanitizer link options
if("${ENABLE_ASAN}" STREQUAL "on")
    add_link_options(-fsanitize=address -fsanitize=leak -fsanitize-recover=address,all -fno-omit-frame-pointer)
    if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        add_link_options(-lasan)
    elseif (CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        add_link_options(-shared-libsan)
    endif ()
endif ()

link_libraries(Secodefuzz ${KUDNN_LIBRARIES} pthread)

# link omp when using clang while link gomp while using gcc
if (${COMP_TOOLCHAIN} STREQUAL "clang")
    link_libraries(omp)
else ()
    link_libraries(gomp)
endif ()

add_executable(codegen_fuzz ${PROJECT_SOURCE_DIR}/src/codegen_fuzz.cpp)
add_executable(eltwise_fuzz ${PROJECT_SOURCE_DIR}/src/eltwise_fuzz.cpp)
add_executable(softmax_fuzz ${PROJECT_SOURCE_DIR}/src/softmax_fuzz.cpp)
add_executable(binary_fuzz ${PROJECT_SOURCE_DIR}/src/binary_fuzz.cpp)
add_executable(prelu_fuzz ${PROJECT_SOURCE_DIR}/src/prelu_fuzz.cpp)
add_executable(matmul_fuzz ${PROJECT_SOURCE_DIR}/src/matmul_fuzz.cpp)
add_executable(reduction_fuzz ${PROJECT_SOURCE_DIR}/src/reduction_fuzz.cpp)
add_executable(inner_product_fuzz ${PROJECT_SOURCE_DIR}/src/inner_product_fuzz.cpp)
add_executable(lnorm_fuzz ${PROJECT_SOURCE_DIR}/src/lnorm_fuzz.cpp)
add_executable(bnorm_fuzz ${PROJECT_SOURCE_DIR}/src/bnorm_fuzz.cpp)
add_executable(gnorm_fuzz ${PROJECT_SOURCE_DIR}/src/gnorm_fuzz.cpp)
add_executable(pool_fuzz ${PROJECT_SOURCE_DIR}/src/pool_fuzz.cpp)
add_executable(convolution_fuzz ${PROJECT_SOURCE_DIR}/src/convolution_fuzz.cpp)
add_executable(deconvolution_fuzz ${PROJECT_SOURCE_DIR}/src/deconvolution_fuzz.cpp)
add_executable(sum_fuzz ${PROJECT_SOURCE_DIR}/src/sum_fuzz.cpp)
add_executable(tensor_fuzz ${PROJECT_SOURCE_DIR}/src/tensor_fuzz.cpp)
add_executable(reorder_fuzz ${PROJECT_SOURCE_DIR}/src/reorder_fuzz.cpp)
add_executable(shuffle_fuzz ${PROJECT_SOURCE_DIR}/src/shuffle_fuzz.cpp)
add_executable(concat_fuzz ${PROJECT_SOURCE_DIR}/src/concat_fuzz.cpp)
add_executable(resampling_fuzz ${PROJECT_SOURCE_DIR}/src/resampling_fuzz.cpp)
add_executable(lrn_fuzz ${PROJECT_SOURCE_DIR}/src/lrn_fuzz.cpp)
add_executable(rnn_fuzz ${PROJECT_SOURCE_DIR}/src/rnn_fuzz.cpp)
add_executable(post_ops_fuzz ${PROJECT_SOURCE_DIR}/src/post_ops_fuzz.cpp)
